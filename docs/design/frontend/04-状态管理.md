## 1. Pinia Store

### Task Store

```typescript
// stores/task.ts
import { defineStore } from 'pinia'

export const useTaskStore = defineStore('task', {
  state: () => ({
    tasks: [] as Task[],
    loading: false
  }),
  
  actions: {
    async fetchTasks() {
      this.loading = true
      const res = await api.getTasks()
      this.tasks = res.data
      this.loading = false
    },
    
    async createTask(task: CreateTaskRequest) {
      const res = await api.createTask(task)
      this.tasks.push(res.data)
      return res.data
    }
  }
})
```
