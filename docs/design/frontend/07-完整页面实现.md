## 1. 概览页面 (Dashboard)

```vue
<!-- src/views/Dashboard.vue -->
<template>
  <div class="dashboard">
    <!-- 统计卡片 -->
    <el-row :gutter="20">
      <el-col :span="6">
        <stat-card label="活跃任务" :value="stats.activeTasks" icon="el-icon-document" />
      </el-col>
      <el-col :span="6">
        <stat-card label="总流量" :value="formatBps(stats.totalBps)" icon="el-icon-upload" />
      </el-col>
      <el-col :span="6">
        <stat-card label="报文数" :value="formatNumber(stats.totalPackets)" icon="el-icon-data-line" />
      </el-col>
      <el-col :span="6">
        <stat-card label="CPU使用" :value="stats.cpuUsage + '%'" icon="el-icon-cpu" />
      </el-col>
    </el-row>

    <!-- 实时图表 -->
    <el-card class="chart-card">
      <template #header>实时流量</template>
      <realtime-chart :data="chartData" />
    </el-card>

    <!-- 活跃任务 -->
    <el-card>
      <template #header>活跃任务</template>
      <task-table :tasks="activeTasks" @start="handleStart" @stop="handleStop" />
    </el-card>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import { useSystemStore } from '@/stores/system'

const systemStore = useSystemStore()
const stats = ref({ activeTasks: 0, totalBps: 0, totalPackets: 0, cpuUsage: 0 })
const chartData = ref([])
const activeTasks = ref([])

let ws: WebSocket | null = null

onMounted(() => {
  fetchStats()
  connectWebSocket()
  const timer = setInterval(fetchStats, 5000)
  onUnmounted(() => clearInterval(timer))
})

const connectWebSocket = () => {
  ws = new WebSocket('ws://localhost:8080/ws/metrics')
  ws.onmessage = (event) => {
    const data = JSON.parse(event.data)
    chartData.value.push({ time: Date.now(), value: data.bps })
    if (chartData.value.length > 60) chartData.value.shift()
  }
}
</script>
```

## 2. 任务管理页面

```vue
<!-- src/views/Tasks.vue -->
<template>
  <div class="tasks-page">
    <!-- 操作栏 -->
    <div class="toolbar">
      <el-button type="primary" @click="showCreateDialog = true">
        <el-icon><Plus /></el-icon> 创建任务
      </el-button>
      <el-input v-model="searchText" placeholder="搜索任务" style="width: 300px" />
      <el-select v-model="statusFilter" placeholder="状态筛选">
        <el-option label="全部" value="" />
        <el-option label="运行中" value="running" />
        <el-option label="已完成" value="completed" />
      </el-select>
    </div>

    <!-- 任务表格 -->
    <el-table :data="filteredTasks" v-loading="loading">
      <el-table-column prop="name" label="任务名称" />
      <el-table-column prop="protocol" label="协议" width="100" />
      <el-table-column prop="status" label="状态" width="100">
        <template #default="{ row }">
          <el-tag :type="getStatusType(row.status)">{{ row.status }}</el-tag>
        </template>
      </el-table-column>
      <el-table-column prop="progress" label="进度" width="150">
        <template #default="{ row }">
          <el-progress :percentage="row.progress" />
        </template>
      </el-table-column>
      <el-table-column label="操作" width="200">
        <template #default="{ row }">
          <el-button size="small" @click="handleStart(row.id)" v-if="row.status !== 'running'">
            启动
          </el-button>
          <el-button size="small" @click="handleStop(row.id)" v-if="row.status === 'running'">
            停止
          </el-button>
          <el-button size="small" @click="showDetail(row)">详情</el-button>
          <el-button size="small" type="danger" @click="handleDelete(row.id)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <!-- 分页 -->
    <el-pagination
      v-model:current-page="currentPage"
      v-model:page-size="pageSize"
      :total="total"
      @current-change="fetchTasks"
    />

    <!-- 创建任务对话框 -->
    <create-task-dialog v-model="showCreateDialog" @success="fetchTasks" />
    
    <!-- 任务详情抽屉 -->
    <task-detail-drawer v-model="showDetailDrawer" :task="selectedTask" />
  </div>
</template>

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useTaskStore } from '@/stores/task'
import { ElMessage, ElMessageBox } from 'element-plus'

const taskStore = useTaskStore()
const loading = ref(false)
const searchText = ref('')
const statusFilter = ref('')
const currentPage = ref(1)
const pageSize = ref(20)
const total = ref(0)
const showCreateDialog = ref(false)
const showDetailDrawer = ref(false)
const selectedTask = ref(null)

const filteredTasks = computed(() => {
  return taskStore.tasks.filter(task => {
    const matchSearch = task.name.includes(searchText.value)
    const matchStatus = !statusFilter.value || task.status === statusFilter.value
    return matchSearch && matchStatus
  })
})

const fetchTasks = async () => {
  loading.value = true
  await taskStore.fetchTasks({ page: currentPage.value, size: pageSize.value })
  loading.value = false
}

const handleStart = async (id: string) => {
  await taskStore.startTask(id)
  ElMessage.success('任务已启动')
}

const handleStop = async (id: string) => {
  await taskStore.stopTask(id)
  ElMessage.success('任务已停止')
}

const handleDelete = async (id: string) => {
  await ElMessageBox.confirm('确认删除此任务？', '警告', { type: 'warning' })
  await taskStore.deleteTask(id)
  ElMessage.success('删除成功')
  fetchTasks()
}

onMounted(fetchTasks)
</script>
```

## 3. 创建任务对话框

```vue
<!-- src/components/CreateTaskDialog.vue -->
<template>
  <el-dialog v-model="visible" title="创建任务" width="800px">
    <el-form :model="form" :rules="rules" ref="formRef" label-width="100px">
      <el-form-item label="任务名称" prop="name">
        <el-input v-model="form.name" />
      </el-form-item>
      
      <el-form-item label="描述" prop="description">
        <el-input v-model="form.description" type="textarea" />
      </el-form-item>
      
      <el-form-item label="协议类型" prop="protocol">
        <el-select v-model="form.protocol" @change="handleProtocolChange">
          <el-option label="TCP" value="tcp" />
          <el-option label="UDP" value="udp" />
          <el-option label="HTTP" value="http" />
          <el-option label="DNS" value="dns" />
        </el-select>
      </el-form-item>
      
      <!-- TCP配置 -->
      <template v-if="form.protocol === 'tcp'">
        <el-form-item label="源IP">
          <el-input v-model="form.spec.src_ip" placeholder="192.168.1.1" />
        </el-form-item>
        <el-form-item label="目的IP">
          <el-input v-model="form.spec.dst_ip" placeholder="10.0.0.1" />
        </el-form-item>
        <el-form-item label="端口">
          <el-input-number v-model="form.spec.port" :min="1" :max="65535" />
        </el-form-item>
        <el-form-item label="流数量">
          <el-input-number v-model="form.spec.flow_count" :min="1" />
        </el-form-item>
      </template>
      
      <!-- HTTP配置 -->
      <template v-if="form.protocol === 'http'">
        <el-form-item label="URL">
          <el-input v-model="form.spec.url" placeholder="http://example.com/api" />
        </el-form-item>
        <el-form-item label="方法">
          <el-select v-model="form.spec.method">
            <el-option label="GET" value="GET" />
            <el-option label="POST" value="POST" />
            <el-option label="PUT" value="PUT" />
          </el-select>
        </el-form-item>
      </template>
      
      <el-form-item label="速率限制">
        <el-input v-model="form.spec.bps" placeholder="100M" />
      </el-form-item>
    </el-form>
    
    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" @click="handleSubmit" :loading="submitting">创建</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref, watch } from 'vue'
import { taskApi } from '@/api/task'
import { ElMessage } from 'element-plus'

const props = defineProps<{ modelValue: boolean }>()
const emit = defineEmits(['update:modelValue', 'success'])

const visible = ref(props.modelValue)
const submitting = ref(false)
const formRef = ref()

const form = ref({
  name: '',
  description: '',
  protocol: 'tcp',
  spec: {}
})

const rules = {
  name: [{ required: true, message: '请输入任务名称' }],
  protocol: [{ required: true, message: '请选择协议' }]
}

watch(() => props.modelValue, (val) => { visible.value = val })
watch(visible, (val) => { emit('update:modelValue', val) })

const handleSubmit = async () => {
  await formRef.value.validate()
  submitting.value = true
  try {
    await taskApi.create(form.value)
    ElMessage.success('创建成功')
    visible.value = false
    emit('success')
  } finally {
    submitting.value = false
  }
}
</script>
```
