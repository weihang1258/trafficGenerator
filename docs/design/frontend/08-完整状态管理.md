## 1. Task Store完整实现

```typescript
// src/stores/task.ts
import { defineStore } from 'pinia'
import { taskApi } from '@/api/task'

export const useTaskStore = defineStore('task', {
  state: () => ({
    tasks: [] as Task[],
    currentTask: null as Task | null,
    loading: false,
    total: 0,
    filters: {
      status: '',
      protocol: '',
      search: ''
    }
  }),
  
  getters: {
    runningTasks: (state) => state.tasks.filter(t => t.status === 'running'),
    completedTasks: (state) => state.tasks.filter(t => t.status === 'completed'),
    taskById: (state) => (id: string) => state.tasks.find(t => t.id === id)
  },
  
  actions: {
    async fetchTasks(params?: { page?: number; size?: number }) {
      this.loading = true
      try {
        const { tasks, total } = await taskApi.list({
          page: params?.page || 1,
          size: params?.size || 20,
          ...this.filters
        })
        this.tasks = tasks
        this.total = total
      } finally {
        this.loading = false
      }
    },
    
    async createTask(data: CreateTaskRequest) {
      const task = await taskApi.create(data)
      this.tasks.unshift(task)
      return task
    },
    
    async startTask(id: string) {
      await taskApi.start(id)
      const task = this.taskById(id)
      if (task) task.status = 'running'
    },
    
    async stopTask(id: string) {
      await taskApi.stop(id)
      const task = this.taskById(id)
      if (task) task.status = 'stopped'
    },
    
    async deleteTask(id: string) {
      await taskApi.delete(id)
      this.tasks = this.tasks.filter(t => t.id !== id)
    },
    
    updateTaskStatus(id: string, status: TaskStatus) {
      const task = this.taskById(id)
      if (task) {
        task.status = status.status
        task.progress = status.progress
        task.stats = status.stats
      }
    },
    
    setFilters(filters: Partial<typeof this.filters>) {
      this.filters = { ...this.filters, ...filters }
    }
  }
})
```

// src/stores/strategy.ts
export const useStrategyStore = defineStore('strategy', {
  state: () => ({
    strategies: [] as Strategy[],
    loading: false,
    currentStrategy: null as Strategy | null
  }),
  
  getters: {
    strategiesByProtocol: (state) => (protocol: string) => 
      state.strategies.filter(s => s.protocol === protocol),
    
    strategyById: (state) => (id: string) =>
      state.strategies.find(s => s.id === id)
  },
  
  actions: {
    async fetchStrategies() {
      this.loading = true
      try {
        const { strategies } = await strategyApi.list()
        this.strategies = strategies
      } finally {
        this.loading = false
      }
    },
    
    async createStrategy(data: CreateStrategyRequest) {
      const strategy = await strategyApi.create(data)
      this.strategies.push(strategy)
      return strategy
    },
    
    async updateStrategy(id: string, data: UpdateStrategyRequest) {
      const strategy = await strategyApi.update(id, data)
      const index = this.strategies.findIndex(s => s.id === id)
      if (index !== -1) {
        this.strategies[index] = strategy
      }
    },
    
    async deleteStrategy(id: string) {
      await strategyApi.delete(id)
      this.strategies = this.strategies.filter(s => s.id !== id)
    }
  }
})

// src/stores/system.ts
export const useSystemStore = defineStore('system', {
  state: () => ({
    status: null as SystemStatus | null,
    metrics: null as Metrics | null,
    protocols: [] as string[],
    loading: false
  }),
  
  getters: {
    isHealthy: (state) => state.status?.running === true,
    cpuUsage: (state) => state.status?.cpu_usage || 0,
    memoryUsage: (state) => state.status?.memory_mb || 0
  },
  
  actions: {
    async fetchStatus() {
      const status = await systemApi.getStatus()
      this.status = status
    },
    
    async fetchMetrics() {
      const metrics = await systemApi.getMetrics()
      this.metrics = metrics
    },
    
    async fetchProtocols() {
      const protocols = await systemApi.getProtocols()
      this.protocols = protocols
    },
    
    startMetricsPolling(interval = 5000) {
      const timer = setInterval(() => {
        this.fetchMetrics()
      }, interval)
      
      return () => clearInterval(timer)
    }
  }
})

## 4. WebSocket Store

```typescript
// src/stores/websocket.ts
export const useWebSocketStore = defineStore('websocket', {
  state: () => ({
    connections: new Map<string, WebSocket>(),
    connected: false
  }),
  
  actions: {
    connectTask(taskId: string, onMessage: (data: any) => void) {
      if (this.connections.has(taskId)) {
        return
      }
      
      const ws = new WebSocket(`ws://localhost:8080/ws/tasks/${taskId}`)
      
      ws.onopen = () => {
        this.connected = true
      }
      
      ws.onmessage = (event) => {
        const data = JSON.parse(event.data)
        onMessage(data)
      }
      
      ws.onclose = () => {
        this.connections.delete(taskId)
        this.connected = false
      }
      
      this.connections.set(taskId, ws)
    },
    
    disconnect(taskId: string) {
      const ws = this.connections.get(taskId)
      if (ws) {
        ws.close()
        this.connections.delete(taskId)
      }
    },
    
    disconnectAll() {
      this.connections.forEach(ws => ws.close())
      this.connections.clear()
    }
  }
})
```

## 5. 组合使用

```vue
<script setup lang="ts">
import { useTaskStore } from '@/stores/task'
import { useWebSocketStore } from '@/stores/websocket'
import { onMounted, onUnmounted } from 'vue'

const taskStore = useTaskStore()
const wsStore = useWebSocketStore()

onMounted(() => {
  taskStore.fetchTasks()
  
  // 为运行中的任务建立WebSocket连接
  taskStore.runningTasks.forEach(task => {
    wsStore.connectTask(task.id, (data) => {
      taskStore.updateTaskStatus(task.id, data)
    })
  })
})

onUnmounted(() => {
  wsStore.disconnectAll()
})
</script>
```
