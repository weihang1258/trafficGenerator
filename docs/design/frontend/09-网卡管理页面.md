## 2. API定义

```typescript
// src/api/interface.ts
export const interfaceApi = {
  list() {
    return request.get('/interfaces')
  },
  
  discover() {
    return request.post('/interfaces/discover')
  },
  
  getPortStatus() {
    return request.get('/ports/status')
  }
}
```
