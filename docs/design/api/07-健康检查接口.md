## 1. 健康检查端点

### 1.1 Liveness探针

```go
// GET /health
// 用途: K8s liveness probe，检查服务是否存活

func healthHandler(c *gin.Context) {
    c.JSON(200, gin.H{
        "status": "ok",
        "timestamp": time.Now().Unix(),
    })
}
```

### 1.2 Readiness探针

```go
// GET /ready
// 用途: K8s readiness probe，检查服务是否就绪

func readyHandler(c *gin.Context) {
    // 检查依赖服务
    checks := map[string]bool{
        "engine":   engine.IsRunning(),
        "database": db.Ping() == nil,
        "redis":    redis.Ping() == nil,
    }
    
    allReady := true
    for _, ready := range checks {
        if !ready {
            allReady = false
            break
        }
    }
    
    if allReady {
        c.JSON(200, gin.H{
            "status": "ready",
            "checks": checks,
        })
    } else {
        c.JSON(503, gin.H{
            "status": "not_ready",
            "checks": checks,
        })
    }
}
```

## 2. 路由注册

```go
// internal/api/rest/router.go
func setupRouter(engine *core.Engine, db *sql.DB) *gin.Engine {
    r := gin.Default()
    
    // 健康检查（无需认证）
    r.GET("/health", healthHandler)
    r.GET("/ready", readyHandler)
    
    // API路由
    v1 := r.Group("/api/v1")
    v1.Use(AuthMiddleware())
    {
        // 任务管理
        v1.POST("/tasks", createTask)
        v1.GET("/tasks", listTasks)
        // ...
    }
    
    return r
}
```
