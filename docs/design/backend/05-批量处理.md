## 1. 批量执行器

```go
// pkg/batch/runner.go
type BatchRunner struct {
    engine  *core.Engine
    spec    BatchSpec
    workers []*ClassWorker
    stop    chan struct{}
}

func (br *BatchRunner) Start() {
    for _, class := range br.spec.Classes {
        worker := &ClassWorker{
            class:  class,
            engine: br.engine,
            stop:   br.stop,
        }
        go worker.Run()
        br.workers = append(br.workers, worker)
    }
}
```

## 2. 类别Worker

```go
type ClassWorker struct {
    class  TrafficClass
    engine *core.Engine
    stop   chan struct{}
}

func (cw *ClassWorker) Run() {
    tupleGen := generators.NewTupleGenerator(cw.class.Tuples)
    
    for {
        select {
        case <-cw.stop:
            return
        default:
            tuple := tupleGen.Next()
            task := buildTask(cw.class, tuple)
            cw.engine.SubmitTask(task)
            
            // 速率控制
            if cw.class.FlowsPerSecond > 0 {
                time.Sleep(time.Second / time.Duration(cw.class.FlowsPerSecond))
            }
        }
    }
}
```
