## 1. 批量执行器完整实现

```go
// pkg/batch/runner.go
package batch

import (
    "context"
    "sync"
    "time"
)

type BatchRunner struct {
    engine    *core.Engine
    spec      BatchSpec
    workers   []*ClassWorker
    ctx       context.Context
    cancel    context.CancelFunc
    wg        sync.WaitGroup
    stats     *BatchStats
    mu        sync.RWMutex
}

type BatchStats struct {
    TotalFlows      int64
    TotalPackets    int64
    TotalBytes      int64
    CurrentBPS      float64
    StartTime       time.Time
    ClassStats      map[string]*ClassStats
}

type ClassStats struct {
    FlowsGenerated  int64
    PacketsSent     int64
    BytesSent       int64
    CurrentBPS      float64
}

func NewBatchRunner(engine *core.Engine, spec BatchSpec) *BatchRunner {
    ctx, cancel := context.WithCancel(context.Background())
    
    return &BatchRunner{
        engine: engine,
        spec:   spec,
        ctx:    ctx,
        cancel: cancel,
        stats: &BatchStats{
            StartTime:  time.Now(),
            ClassStats: make(map[string]*ClassStats),
        },
    }
}

func (br *BatchRunner) Start() error {
    for _, class := range br.spec.Classes {
        // 设置速率限制
        if class.BPS != "" {
            bps := parseBPS(class.BPS)
            br.engine.SetClassRateLimit(class.ID, bps)
        }
        
        // 创建类别Worker
        worker := NewClassWorker(br.engine, class, br.stats)
        br.workers = append(br.workers, worker)
        
        // 启动Worker
        br.wg.Add(1)
        go func(w *ClassWorker) {
            defer br.wg.Done()
            w.Run(br.ctx)
        }(worker)
    }
    
    // 启动统计收集
    br.wg.Add(1)
    go br.collectStats()
    
    return nil
}

func (br *BatchRunner) Stop() error {
    br.cancel()
    br.wg.Wait()
    return nil
}

func (br *BatchRunner) GetStatus() BatchStatus {
    br.mu.RLock()
    defer br.mu.RUnlock()
    
    return BatchStatus{
        Running:      br.ctx.Err() == nil,
        TotalFlows:   br.stats.TotalFlows,
        TotalPackets: br.stats.TotalPackets,
        CurrentBPS:   br.stats.CurrentBPS,
        Duration:     time.Since(br.stats.StartTime),
        ClassStats:   br.stats.ClassStats,
    }
}

func (br *BatchRunner) collectStats() {
    defer br.wg.Done()
    
    ticker := time.NewTicker(time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ticker.C:
            br.updateStats()
        case <-br.ctx.Done():
            return
        }
    }
}

func (br *BatchRunner) updateStats() {
    br.mu.Lock()
    defer br.mu.Unlock()
    
    var totalBPS float64
    for _, worker := range br.workers {
        stats := worker.GetStats()
        br.stats.ClassStats[worker.class.ID] = stats
        totalBPS += stats.CurrentBPS
    }
    br.stats.CurrentBPS = totalBPS
}
```

## 2. 类别Worker完整实现

```go
// pkg/batch/class_worker.go
type ClassWorker struct {
    engine      *core.Engine
    class       TrafficClass
    batchStats  *BatchStats
    tupleGen    *generators.TupleGenerator
    bodyGen     *generators.BodyGenerator
    flowCount   int64
    packetCount int64
    byteCount   int64
    lastUpdate  time.Time
    mu          sync.Mutex
}

func NewClassWorker(engine *core.Engine, class TrafficClass, stats *BatchStats) *ClassWorker {
    return &ClassWorker{
        engine:     engine,
        class:      class,
        batchStats: stats,
        tupleGen:   generators.NewTupleGenerator(class.Tuples),
        bodyGen:    generators.NewBodyGenerator(class.Body),
        lastUpdate: time.Now(),
    }
}

func (cw *ClassWorker) Run(ctx context.Context) {
    flowsPerSecond := cw.class.FlowsPerSecond
    if flowsPerSecond == 0 {
        flowsPerSecond = 100 // 默认值
    }
    
    interval := time.Second / time.Duration(flowsPerSecond)
    ticker := time.NewTicker(interval)
    defer ticker.Stop()
    
    maxFlows := cw.class.FlowCount
    
    for {
        select {
        case <-ticker.C:
            if maxFlows > 0 && cw.flowCount >= int64(maxFlows) {
                return
            }
            
            if err := cw.generateFlow(); err != nil {
                log.Error("generate flow failed", zap.Error(err))
                continue
            }
            
            atomic.AddInt64(&cw.flowCount, 1)
            
        case <-ctx.Done():
            return
        }
    }
}

func (cw *ClassWorker) generateFlow() error {
    tuple := cw.tupleGen.Next()
    
    task := Task{
        ID:       generateTaskID(),
        Protocol: cw.class.Type,
        Spec: FlowSpec{
            SrcIP:   tuple.SrcIP,
            DstIP:   tuple.DstIP,
            SrcPort: tuple.SrcPort,
            DstPort: tuple.DstPort,
            Body:    cw.bodyGen.Next(),
        },
        ClassID: cw.class.ID,
    }
    
    return cw.engine.SubmitTask(task)
}

func (cw *ClassWorker) GetStats() *ClassStats {
    cw.mu.Lock()
    defer cw.mu.Unlock()
    
    now := time.Now()
    duration := now.Sub(cw.lastUpdate).Seconds()
    
    stats := &ClassStats{
        FlowsGenerated: cw.flowCount,
        PacketsSent:    cw.packetCount,
        BytesSent:      cw.byteCount,
        CurrentBPS:     float64(cw.byteCount) * 8 / duration,
    }
    
    cw.lastUpdate = now
    return stats
}
```

## 3. 使用示例

```go
// 创建批量规范
spec := BatchSpec{
    Classes: []TrafficClass{
        {
            ID:             "tcp_web",
            Type:           "tcp",
            BPS:            "200M",
            FlowsPerSecond: 100,
            FlowCount:      10000,
            Tuples: TupleConfig{
                SrcIP:   StrategyConfig{Strategy: "inc", Range: []string{"10.0.1.1", "10.0.1.100"}},
                DstIP:   StrategyConfig{Strategy: "fixed", Value: "192.168.1.1"},
                SrcPort: StrategyConfig{Strategy: "random", Range: []int{10000, 20000}},
                DstPort: StrategyConfig{Strategy: "fixed", Value: 80},
            },
        },
        {
            ID:             "udp_stream",
            Type:           "udp",
            BPS:            "100M",
            FlowsPerSecond: 200,
            FlowCount:      5000,
        },
    },
}

// 启动批量生成
runner := NewBatchRunner(engine, spec)
runner.Start()

// 监控状态
go func() {
    for {
        status := runner.GetStatus()
        log.Info("batch status", 
            zap.Int64("flows", status.TotalFlows),
            zap.Float64("bps", status.CurrentBPS))
        time.Sleep(5 * time.Second)
    }
}()

// 停止
time.Sleep(60 * time.Second)
runner.Stop()
```
