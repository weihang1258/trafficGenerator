## 1. 主程序结构

```go
// cmd/server/main.go
package main

import (
    "context"
    "log"
    "os"
    "os/signal"
    "syscall"
    "time"
    
    "github.com/yourorg/traffic-generator/internal/api/rest"
    "github.com/yourorg/traffic-generator/internal/api/grpc"
    "github.com/yourorg/traffic-generator/internal/config"
    "github.com/yourorg/traffic-generator/internal/storage"
    "github.com/yourorg/traffic-generator/pkg/core"
    "go.uber.org/zap"
)

func main() {
    // 1. 加载配置
    cfg, err := config.Load("configs/config.yaml")
    if err != nil {
        log.Fatal("load config failed:", err)
    }
    
    // 2. 初始化日志
    logger := initLogger(cfg.Logging)
    defer logger.Sync()
    
    // 3. 初始化数据库
    db, err := storage.NewDB(cfg.Database)
    if err != nil {
        logger.Fatal("init database failed", zap.Error(err))
    }
    defer db.Close()
    
    // 4. 运行迁移
    if err := storage.RunMigrations(db); err != nil {
        logger.Fatal("run migrations failed", zap.Error(err))
    }
    
    // 5. 初始化Redis
    var redisClient *storage.RedisClient
    if cfg.Redis.Enabled {
        redisClient = storage.NewRedisClient(cfg.Redis)
        defer redisClient.Close()
    }
    
    // 6. 创建核心引擎
    engine := core.NewEngine(&core.EngineConfig{
        ConfigWorkers: cfg.Engine.ConfigWorkers,
        PacketWorkers: cfg.Engine.PacketWorkers,
        BufferSize:    cfg.Engine.BufferSize,
        Logger:        logger,
    })
    
    // 7. 启动引擎
    if err := engine.Start(); err != nil {
        logger.Fatal("start engine failed", zap.Error(err))
    }
    defer engine.Stop()
    
    // 8. 启动API服务
    restServer := rest.NewServer(cfg.Server, engine, db, logger)
    grpcServer := grpc.NewServer(cfg.Server, engine, logger)
    
    go func() {
        if err := restServer.Start(); err != nil {
            logger.Fatal("rest server failed", zap.Error(err))
        }
    }()
    
    go func() {
        if err := grpcServer.Start(); err != nil {
            logger.Fatal("grpc server failed", zap.Error(err))
        }
    }()
    
    // 9. 启动监控
    if cfg.Metrics.Enabled {
        go startMetricsServer(cfg.Metrics, logger)
    }
    
    logger.Info("traffic generator started",
        zap.Int("rest_port", cfg.Server.RestPort),
        zap.Int("grpc_port", cfg.Server.GrpcPort))
    
    // 10. 优雅关闭
    waitForShutdown(logger, engine, restServer, grpcServer)
}
```

## 2. 依赖注入

```go
// internal/app/app.go
package app

type App struct {
    Config      *config.Config
    Logger      *zap.Logger
    DB          *sql.DB
    Redis       *storage.RedisClient
    Engine      *core.Engine
    RestServer  *rest.Server
    GrpcServer  *grpc.Server
}

func New(configPath string) (*App, error) {
    app := &App{}
    
    // 加载配置
    cfg, err := config.Load(configPath)
    if err != nil {
        return nil, err
    }
    app.Config = cfg
    
    // 初始化日志
    app.Logger = initLogger(cfg.Logging)
    
    // 初始化数据库
    db, err := storage.NewDB(cfg.Database)
    if err != nil {
        return nil, err
    }
    app.DB = db
    
    // 初始化引擎
    app.Engine = core.NewEngine(&core.EngineConfig{
        ConfigWorkers: cfg.Engine.ConfigWorkers,
        PacketWorkers: cfg.Engine.PacketWorkers,
        Logger:        app.Logger,
    })
    
    // 初始化服务
    app.RestServer = rest.NewServer(cfg.Server, app.Engine, app.DB, app.Logger)
    app.GrpcServer = grpc.NewServer(cfg.Server, app.Engine, app.Logger)
    
    return app, nil
}

func (a *App) Start() error {
    if err := a.Engine.Start(); err != nil {
        return err
    }
    
    go a.RestServer.Start()
    go a.GrpcServer.Start()
    
    return nil
}

func (a *App) Stop() error {
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    a.RestServer.Shutdown(ctx)
    a.GrpcServer.Stop()
    a.Engine.Stop()
    a.DB.Close()
    
    return nil
}
```

## 3. 优雅关闭

```go
func waitForShutdown(logger *zap.Logger, components ...interface{ Stop() error }) {
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
    
    sig := <-sigChan
    logger.Info("received shutdown signal", zap.String("signal", sig.String()))
    
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    done := make(chan struct{})
    go func() {
        for _, comp := range components {
            if err := comp.Stop(); err != nil {
                logger.Error("stop component failed", zap.Error(err))
            }
        }
        close(done)
    }()
    
    select {
    case <-done:
        logger.Info("graceful shutdown completed")
    case <-ctx.Done():
        logger.Warn("shutdown timeout, force exit")
    }
}
```

## 4. 启动顺序

```
1. 加载配置 (config.yaml + 环境变量)
   ↓
2. 初始化日志 (zap)
   ↓
3. 连接数据库 (SQLite/PostgreSQL)
   ↓
4. 运行数据库迁移
   ↓
5. 连接Redis (可选)
   ↓
6. 创建核心引擎
   ↓
7. 启动引擎 (Worker池)
   ↓
8. 启动REST API服务器
   ↓
9. 启动gRPC服务器
   ↓
10. 启动Prometheus监控
   ↓
11. 等待关闭信号
```

## 5. 命令行参数

```go
import "github.com/spf13/cobra"

var rootCmd = &cobra.Command{
    Use:   "traffic-generator",
    Short: "High-performance traffic generator",
}

var serverCmd = &cobra.Command{
    Use:   "server",
    Short: "Start the server",
    Run: func(cmd *cobra.Command, args []string) {
        configPath, _ := cmd.Flags().GetString("config")
        main(configPath)
    },
}

func init() {
    serverCmd.Flags().StringP("config", "c", "configs/config.yaml", "config file path")
    rootCmd.AddCommand(serverCmd)
}

func Execute() {
    if err := rootCmd.Execute(); err != nil {
        os.Exit(1)
    }
}
```

使用:
```bash
./traffic-generator server --config configs/config.prod.yaml
```
