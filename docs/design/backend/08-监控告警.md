## 2. 告警规则

### 2.1 Prometheus告警配置

```yaml
groups:
  - name: traffic_generator_alerts
    interval: 30s
    rules:
      # CPU告警
      - alert: HighCPUUsage
        expr: process_cpu_seconds_total > 0.8
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "CPU使用率过高"
          description: "CPU使用率超过80%持续5分钟"
      
      # 内存告警
      - alert: HighMemoryUsage
        expr: process_resident_memory_bytes > 2e9
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "内存使用过高"
          description: "内存使用超过2GB"
      
      # 缓冲区告警
      - alert: BufferNearFull
        expr: tg_buffer_usage_percent > 90
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "缓冲区接近满"
          description: "{{ $labels.buffer_type }}缓冲区使用率>90%"
      
      # 错误率告警
      - alert: HighErrorRate
        expr: rate(tg_errors_total[5m]) > 10
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "错误率过高"
          description: "{{ $labels.component }}错误率>10/s"
      
      # 任务失败告警
      - alert: TaskFailures
        expr: increase(tg_tasks_total{status="failed"}[5m]) > 5
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "任务失败过多"
          description: "5分钟内失败任务>5个"
      
      # 性能下降告警
      - alert: LowPacketRate
        expr: tg_packet_rate_pps < 100000
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "报文生成速率低"
          description: "报文速率<100K pps持续5分钟"
```

// 日志级别
const (
    LogLevelDebug = "debug"
    LogLevelInfo  = "info"
    LogLevelWarn  = "warn"
    LogLevelError = "error"
    LogLevelFatal = "fatal"
)

// 日志配置
func InitLogger(level string) *zap.Logger {
    config := zap.NewProductionConfig()
    config.Level = zap.NewAtomicLevelAt(parseLevel(level))
    config.OutputPaths = []string{"stdout", "/var/log/tg/app.log"}
    config.ErrorOutputPaths = []string{"stderr", "/var/log/tg/error.log"}
    
    logger, _ := config.Build()
    return logger
}

// 结构化日志
log.Info("task started",
    zap.String("task_id", taskID),
    zap.String("protocol", protocol),

log.Error("packet build failed",

## 4. 监控采集

```go
func (e *Engine) collectMetrics() {
    ticker := time.NewTicker(5 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ticker.C:
            // 缓冲区使用率
            BufferUsage.WithLabelValues("up").Set(
                float64(e.upBuffer.Size()) / float64(e.upBuffer.Capacity()) * 100)
            BufferUsage.WithLabelValues("down").Set(
                float64(e.downBuffer.Size()) / float64(e.downBuffer.Capacity()) * 100)
            
            // Worker利用率
            WorkerUtilization.WithLabelValues("config").Set(
                e.calcWorkerUtilization("config"))
            WorkerUtilization.WithLabelValues("packet").Set(
                e.calcWorkerUtilization("packet"))
            
        case <-e.shutdownCh:
            return
        }
    }
}
```

## 5. 链路追踪

```go
import "go.opentelemetry.io/otel/trace"

func (e *Engine) SubmitTaskWithTrace(ctx context.Context, task Task) error {
    ctx, span := tracer.Start(ctx, "Engine.SubmitTask")
    defer span.End()
    
    span.SetAttributes(
        attribute.String("task.id", task.ID),
        attribute.String("task.protocol", task.Protocol))
    
    if err := e.submitTask(ctx, task); err != nil {
        span.RecordError(err)
        return err
    }
    
    return nil
}
```
