## 1. 完整配置示例

```yaml
# configs/config.yaml

# 服务配置
server:
  host: "0.0.0.0"
  rest_port: 8080
  grpc_port: 9090
  websocket_port: 8081
  mode: "release"  # debug, release
  
# 引擎配置
engine:
  config_workers: 4      # 配置生成Worker数
  packet_workers: 4      # 报文构造Worker数
  buffer_size: 10000     # 环形缓冲区大小
  channel_size: 2000     # Channel缓冲大小
  enable_up_buffer: true
  enable_down_buffer: true
  enable_combined_buffer: true

# 输出配置
output:
  mode: "interface"      # interface, pcap, both, callback
  interfaces:            # 网卡列表
    - "eth0"
  pcap_file: "/tmp/traffic.pcap"  # PCAP文件路径
  batch_size: 100        # 批量发送大小
  buffer_flush_ms: 10    # 缓冲刷新间隔(毫秒)

# 数据库配置
database:
  type: "sqlite"  # sqlite, postgres
  sqlite:
    path: "./data/traffic_generator.db"
  postgres:
    host: "localhost"
    port: 5432
    user: "admin"
    password: "password"
    dbname: "traffic_generator"
    sslmode: "disable"
    max_open_conns: 25
    max_idle_conns: 5

# Redis配置
redis:
  enabled: true
  host: "localhost"
  port: 6379
  password: ""
  db: 0
  pool_size: 10

# 日志配置
logging:
  level: "info"  # debug, info, warn, error
  format: "json"  # json, console
  output:
    - "stdout"
    - "/var/log/traffic-generator/app.log"
  error_output:
    - "stderr"
    - "/var/log/traffic-generator/error.log"
  max_size: 100      # MB
  max_backups: 10
  max_age: 30        # days
  compress: true

# 监控配置
metrics:
  enabled: true
  port: 9091
  path: "/metrics"
  
# 认证配置
auth:
  enabled: true
  jwt_secret: "${JWT_SECRET}"  # 从环境变量读取
  token_expire: 24h
  
# 限流配置
rate_limit:
  enabled: true
  requests_per_second: 100
  burst: 200
  
# CORS配置
cors:
  enabled: true
  allowed_origins:
    - "http://localhost:3000"
    - "https://yourdomain.com"
  allowed_methods:
    - "GET"
    - "POST"
    - "PUT"
    - "DELETE"
  allowed_headers:
    - "Authorization"
    - "Content-Type"
```

## 2. 环境变量覆盖

支持通过环境变量覆盖配置：

```bash
# 格式: TG_<SECTION>_<KEY>
export TG_SERVER_REST_PORT=8888
export TG_DATABASE_TYPE=postgres
export TG_LOGGING_LEVEL=debug
export JWT_SECRET=your-secret-key
```

优先级: **环境变量 > 配置文件 > 默认值**

## 3. 配置加载代码

```go
// internal/config/config.go
package config

import (
    "github.com/spf13/viper"
)

type Config struct {
    Server   ServerConfig   `mapstructure:"server"`
    Engine   EngineConfig   `mapstructure:"engine"`
    Database DatabaseConfig `mapstructure:"database"`
    Redis    RedisConfig    `mapstructure:"redis"`
    Logging  LoggingConfig  `mapstructure:"logging"`
    Metrics  MetricsConfig  `mapstructure:"metrics"`
    Auth     AuthConfig     `mapstructure:"auth"`
}

type ServerConfig struct {
    Host          string `mapstructure:"host"`
    RestPort      int    `mapstructure:"rest_port"`
    GrpcPort      int    `mapstructure:"grpc_port"`
    WebsocketPort int    `mapstructure:"websocket_port"`
    Mode          string `mapstructure:"mode"`
}

type EngineConfig struct {
    ConfigWorkers  int  `mapstructure:"config_workers"`
    PacketWorkers  int  `mapstructure:"packet_workers"`
    BufferSize     int  `mapstructure:"buffer_size"`
    ChannelSize    int  `mapstructure:"channel_size"`
}

func Load(configPath string) (*Config, error) {
    viper.SetConfigFile(configPath)
    viper.SetEnvPrefix("TG")
    viper.AutomaticEnv()
    
    // 默认值
    viper.SetDefault("server.rest_port", 8080)
    viper.SetDefault("engine.config_workers", 4)
    viper.SetDefault("logging.level", "info")
    
    if err := viper.ReadInConfig(); err != nil {
        return nil, err
    }
    
    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, err
    }
    
    return &cfg, nil
}
```

## 4. 使用示例

```go
// cmd/server/main.go
cfg, err := config.Load("configs/config.yaml")
if err != nil {
    log.Fatal(err)
}

// 使用配置
engine := core.NewEngine(&core.EngineConfig{
    ConfigWorkers: cfg.Engine.ConfigWorkers,
    PacketWorkers: cfg.Engine.PacketWorkers,
    BufferSize:    cfg.Engine.BufferSize,
})
```

## 5. 不同环境配置

```
configs/
├── config.yaml          # 默认配置
├── config.dev.yaml      # 开发环境
├── config.prod.yaml     # 生产环境
└── config.test.yaml     # 测试环境
```

启动时指定:
```bash
./traffic-generator server --config configs/config.prod.yaml
```
