## 1. 配置模板

系统提供多种部署场景的配置模板：

| 模板 | 文件 | 用途 |
|------|------|------|
| 开发环境 | `config.dev.yaml` | debug 模式，console 日志，无 Redis |
| 独立部署 | `config.standalone.yaml` | SQLite，无 Redis，环境变量替换 |
| Docker Compose | `config.docker.yaml` | Redis 启用，PostgreSQL 就绪，环境变量替换 |

### 1.1 使用方式

```bash
# 复制模板到 config.yaml
cp configs/config.standalone.yaml configs/config.yaml

# 设置环境变量
export JWT_SECRET="your-secret"
export ADMIN_USERNAME="admin"
export ADMIN_PASSWORD="your-password"
export ADMIN_EMAIL="admin@example.com"

# 启动服务
./bin/trafficgen -config configs/config.yaml
```

**重要**: `config.yaml` 已被 `.gitignore` 忽略，不会同步到 git，避免敏感信息泄露。

## 2. 完整配置示例

```yaml
# configs/config.yaml

# 服务配置
server:
  host: "0.0.0.0"
  port: 8080
  mode: "release"  # debug, release, test
  
# 引擎配置
engine:
  config_workers: 4      # 配置生成Worker数
  packet_workers: 8      # 报文构造Worker数
  output_workers: 4      # 输出Worker数
  buffer_size: 4096      # 环形缓冲区大小
  queue_size: 1024       # 队列大小

# 数据库配置
database:
  type: "sqlite"  # sqlite, postgres
  sqlite:
    path: "./data/trafficgen.db"
  postgres:
    host: "localhost"
    port: 5432
    user: "trafficgen"
    password: "${POSTGRES_PASSWORD}"
    dbname: "trafficgen"
    sslmode: "disable"
  pool:
    max_open: 100
    max_idle: 10
    conn_lifetime: "1h"

# Redis配置
redis:
  enabled: false
  addr: "localhost:6379"
  password: ""
  db: 0
  pool_size: 100

# 日志配置
logging:
  level: "info"  # debug, info, warn, error
  format: "json"  # json, console
  output: "stdout"  # stdout, file
  file:
    path: "./logs/trafficgen.log"
    max_size: 100      # MB
    max_backups: 5
    max_age: 30        # days
    compress: true

# 认证配置
auth:
  jwt_secret: "${JWT_SECRET}"
  jwt_issuer: "trafficgen"
  jwt_expires_in: 24  # hours
  admin:
    username: "${ADMIN_USERNAME}"
    password: "${ADMIN_PASSWORD}"
    email: "${ADMIN_EMAIL}"

# 限流配置
rate_limit:
  enabled: true
  requests_per_second: 100
  burst: 200

# 监控配置
metrics:
  enabled: true
  path: "/metrics"
```

## 3. 环境变量支持

### 3.1 环境变量替换

配置文件支持 `${VAR_NAME}` 语法，启动时自动替换为环境变量值：

```yaml
auth:
  jwt_secret: "${JWT_SECRET}"
  admin:
    username: "${ADMIN_USERNAME}"
    password: "${ADMIN_PASSWORD}"
    email: "${ADMIN_EMAIL}"
```

### 3.2 环境变量覆盖

支持通过环境变量覆盖配置：

```bash
# 格式: TG_<SECTION>_<KEY>
export TG_SERVER_PORT=8888
export TG_DATABASE_TYPE=postgres
export TG_LOGGING_LEVEL=debug
export TG_REDIS_ENABLED=true
```

优先级: **环境变量 > 配置文件 > 默认值**

## 4. 配置加载代码

```go
// pkg/config/config.go
package config

import (
    "fmt"
    "os"
    "regexp"
    "time"

    "github.com/spf13/viper"
)

type Config struct {
    Server    ServerConfig    `mapstructure:"server"`
    Database  DatabaseConfig  `mapstructure:"database"`
    Redis     RedisConfig     `mapstructure:"redis"`
    Engine    EngineConfig    `mapstructure:"engine"`
    Logging   LoggingConfig   `mapstructure:"logging"`
    Auth      AuthConfig      `mapstructure:"auth"`
    RateLimit RateLimitConfig `mapstructure:"rate_limit"`
    Metrics   MetricsConfig   `mapstructure:"metrics"`
}

func Load(configPath string) (*Config, error) {
    viper.SetConfigFile(configPath)
    viper.SetEnvPrefix("TG")
    viper.AutomaticEnv()
    
    // 默认值
    setDefaults()
    
    if err := viper.ReadInConfig(); err != nil {
        return nil, err
    }
    
    // 展开环境变量
    expandEnvInViper()
    
    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, err
    }
    
    return &cfg, nil
}

// expandEnv 替换 ${VAR} 为环境变量值
func expandEnv(s string) string {
    re := regexp.MustCompile(`\$\{([^}]+)\}`)
    return re.ReplaceAllStringFunc(s, func(match string) string {
        name := match[2 : len(match)-1]
        return os.Getenv(name)
    })
}
```

## 6. 管理员账户配置

管理员账户通过配置文件初始化，**不能通过 API 或页面创建/修改/删除**：

```yaml
auth:
  admin:
    username: "${ADMIN_USERNAME}"
    password: "${ADMIN_PASSWORD}"
    email: "${ADMIN_EMAIL}"
```

系统启动时会自动创建配置文件中指定的管理员账户。如需修改管理员信息，请直接编辑配置文件并重启服务。
