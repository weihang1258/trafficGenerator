## 2. 基础策略

### Fixed

```go
type FixedStrategy struct {
    value interface{}
}

func (s *FixedStrategy) Next() interface{} {
    return s.value
}
```

### Increment

```go
type IncrementStrategy struct {
    current int
    start   int
    end     int
    step    int
}

func (s *IncrementStrategy) Next() interface{} {
    val := s.current
    s.current += s.step
    if s.current > s.end {
        s.current = s.start
    }
    return val
}
```

### Random

```go
type RandomStrategy struct {
    min int
    max int
    rng *rand.Rand
}

func (s *RandomStrategy) Next() interface{} {
    return s.rng.Intn(s.max-s.min+1) + s.min
}
```

## 3. 四元组生成器

```go
// pkg/generators/tuple.go
type TupleGenerator struct {
    srcIP   Strategy
    dstIP   Strategy
    srcPort Strategy
    dstPort Strategy
}

func (tg *TupleGenerator) Next() Tuple {
    return Tuple{
        SrcIP:   tg.srcIP.Next().(string),
        DstIP:   tg.dstIP.Next().(string),
        SrcPort: tg.srcPort.Next().(uint16),
        DstPort: tg.dstPort.Next().(uint16),
    }
}
```

## 4. Body生成器

```go
// pkg/generators/body.go
type BodyGenerator struct {
    strategy string
    config   map[string]interface{}
}

func (bg *BodyGenerator) Next() []byte {
    switch bg.strategy {
    case "fixed":
        return []byte(bg.config["content"].(string))
    case "random":
        length := bg.config["length"].(int)
        return randomBytes(length)
    case "pattern":
        return []byte(fmt.Sprintf(bg.config["pattern"].(string), 
            bg.config["counter"]))
    }
    return nil
}
```
