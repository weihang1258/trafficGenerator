## 1. 协议注册机制

### 1.1 注册表设计

```go
// pkg/protocols/registry.go
package protocols

type Planner func(spec FlowSpec) <-chan PacketConfig

var registry = make(map[string]Planner)

func Register(name string, planner Planner) {
    registry[name] = planner
}

func Get(name string) Planner {
    return registry[name]
}

func List() []string {
    names := make([]string, 0, len(registry))
    for name := range registry {
        names = append(names, name)
    }
    return names
}
```

## 2. TCP协议

### 2.1 流程设计

```
握手 → 数据传输 → 挥手
SYN → SYN-ACK → ACK → DATA → FIN → FIN-ACK → ACK
```

### 2.2 实现

```go
// pkg/protocols/tcp.go
func PlanTCPFlow(spec FlowSpec) <-chan PacketConfig {
    ch := make(chan PacketConfig, 100)
    
    go func() {
        defer close(ch)
        
        seq := rand.Uint32()
        ack := rand.Uint32()
        
        // 三次握手
        if spec.Handshake {
            ch <- buildSYN(spec, seq)
            ch <- buildSYNACK(spec, ack, seq+1)
            ch <- buildACK(spec, seq+1, ack+1)
            seq++
            ack++
        }
        
        // 数据传输
        for _, segment := range segmentData(spec.Payload, spec.MSS) {
            ch <- buildDataPacket(spec, seq, ack, segment)
            seq += uint32(len(segment))
            ch <- buildACK(spec, ack, seq) // 对端ACK
        }
        
        // 四次挥手
        if spec.Termination {
            ch <- buildFIN(spec, seq, ack)
            ch <- buildFINACK(spec, ack, seq+1)
            ch <- buildACK(spec, seq+1, ack+1)
        }
    }()
    
    return ch
}
```

## 3. UDP协议

### 3.1 实现

```go
// pkg/protocols/udp.go
func PlanUDPFlow(spec FlowSpec) <-chan PacketConfig {
    ch := make(chan PacketConfig, 10)
    
    go func() {
        defer close(ch)
        
        // 请求
        ch <- PacketConfig{
            L2: buildL2(spec),
            L3: buildL3(spec),
            L4: buildUDP(spec.SrcPort, spec.DstPort),
            Payload: spec.Payload,
            Direction: "up",
        }
        
        // 响应
        if spec.Response {
            ch <- PacketConfig{
                L2: buildL2Reverse(spec),
                L3: buildL3Reverse(spec),
                L4: buildUDP(spec.DstPort, spec.SrcPort),
                Payload: spec.ResponsePayload,
                Direction: "down",
            }
        }
    }()
    
    return ch
}
```

## 4. HTTP协议

### 4.1 实现

```go
// pkg/protocols/http.go
func PlanHTTPSession(spec HTTPSpec) <-chan PacketConfig {
    ch := make(chan PacketConfig, 100)
    
    go func() {
        defer close(ch)
        
        // TCP握手
        tcpCh := PlanTCPFlow(spec.TCPSpec)
        for cfg := range tcpCh {
            if cfg.IsSYN() || cfg.IsSYNACK() || cfg.IsHandshakeACK() {
                ch <- cfg
            }
        }
        
        // HTTP事务
        for _, txn := range spec.Transactions {
            // 请求
            reqData := buildHTTPRequest(txn.Method, txn.URI, txn.Headers, txn.Body)
            for cfg := range sendData(spec, reqData, "up") {
                ch <- cfg
            }
            
            // 响应
            respData := buildHTTPResponse(txn.Status, txn.RespHeaders, txn.RespBody)
            for cfg := range sendData(spec, respData, "down") {
                ch <- cfg
            }
            
            // Think time
            if txn.ThinkTime > 0 {
                time.Sleep(txn.ThinkTime)
            }
        }
        
        // TCP挥手
        for cfg := range tcpCh {
            if cfg.IsFIN() {
                ch <- cfg
            }
        }
    }()
    
    return ch
}
```

## 5. DNS协议

### 5.1 实现

```go
// pkg/protocols/dns.go
func PlanDNSFlow(spec DNSSpec) <-chan PacketConfig {
    ch := make(chan PacketConfig, 2)
    
    go func() {
        defer close(ch)
        
        // DNS查询
        query := buildDNSQuery(spec.Domain, spec.Type)
        ch <- PacketConfig{
            L2: buildL2(spec),
            L3: buildL3(spec),
            L4: buildUDP(spec.SrcPort, 53),
            Payload: query,
            Direction: "up",
        }
        
        // DNS响应
        response := buildDNSResponse(spec.Domain, spec.IP)
        ch <- PacketConfig{
            L2: buildL2Reverse(spec),
            L3: buildL3Reverse(spec),
            L4: buildUDP(53, spec.SrcPort),
            Payload: response,
            Direction: "down",
        }
    }()
    
    return ch
}
```
