## 1. 错误处理架构

```
Worker Panic → errChan → Error Handler → Log/Metrics/Shutdown
```

---

## 2. 错误分类

```go
type ErrorLevel int

const (
    ErrorLevelWarning ErrorLevel = iota  // 可恢复
    ErrorLevelError                      // 需记录
    ErrorLevelFatal                      // 需停止
)

type EngineError struct {
    Level     ErrorLevel
    Component string
    Message   string
    Err       error
    Timestamp time.Time
}
```

---

## 3. 统一错误处理

```go
func (e *Engine) handleErrors() {
    for err := range e.errChan {
        engineErr := err.(*EngineError)
        
        // 记录日志
        log.Error("engine error",
            zap.String("component", engineErr.Component),
            zap.Error(engineErr.Err))
        
        // 更新指标
        metrics.ErrorCounter.WithLabelValues(
            engineErr.Component).Inc()
        
        // 致命错误处理
        if engineErr.Level == ErrorLevelFatal {
            e.fatalErr.Store(engineErr)
            e.shutdown()
        }
    }
}
```

## 4. Worker Panic恢复

```go
func workerRecover(id int, component string, errChan chan<- error) {
    if r := recover(); r != nil {
        errChan <- &EngineError{
            Level:     ErrorLevelError,
            Component: component,
            Message:   fmt.Sprintf("worker %d panic", id),
            Err:       fmt.Errorf("%v", r),
            Timestamp: time.Now(),
        }
    }
}
```

## 5. 超时控制

```go
func (e *Engine) SubmitTaskWithTimeout(task Task, timeout time.Duration) error {
    ctx, cancel := context.WithTimeout(context.Background(), timeout)
    defer cancel()
    
    select {
    case e.taskChan <- task:
        return nil
    case <-ctx.Done():
        return fmt.Errorf("submit task timeout: %w", ctx.Err())
    }
}
```

## 6. 优雅关闭

```go
func (e *Engine) Stop() error {
    if !e.running.CompareAndSwap(true, false) {
        return errors.New("engine not running")
    }
    
    // 1. 停止接收新任务
    close(e.shutdownCh)
    
    // 2. 等待Worker完成
    done := make(chan struct{})
    go func() {
        e.wg.Wait()
        close(done)
    }()
    
    // 3. 超时强制退出
    select {
    case <-done:
        log.Info("engine stopped gracefully")
    case <-time.After(30 * time.Second):
        log.Warn("engine stop timeout, force exit")
    }
    
    // 4. 关闭Channel
    close(e.taskChan)
    close(e.configChan)
    close(e.packetChan)
    close(e.errChan)
    
    return nil
}
```

## 7. 错误重试策略

```go
type RetryConfig struct {
    MaxRetries int
    Backoff    time.Duration
}

func retryWithBackoff(fn func() error, cfg RetryConfig) error {
    var err error
    for i := 0; i < cfg.MaxRetries; i++ {
        if err = fn(); err == nil {
            return nil
        }
        time.Sleep(cfg.Backoff * time.Duration(i+1))
    }
    return fmt.Errorf("max retries exceeded: %w", err)
}
```
