## 3. Worker池设计

### 3.1 Config Worker

**职责**：消费任务，生成报文配置

```go
func configWorker(
    id int,
    taskChan <-chan Task,
    configChan chan<- PacketConfig,
    errChan chan<- error,
    stopChan <-chan struct{},
) {
    defer func() {
        if r := recover(); r != nil {
            errChan <- fmt.Errorf("config worker %d panic: %v", id, r)
        }
    }()
    
    for {
        select {
        case task, ok := <-taskChan:
            if !ok {
                return
            }
            planner := protocols.Get(task.Protocol)
            if planner == nil {
                errChan <- fmt.Errorf("unknown protocol: %s", task.Protocol)
                continue
            }
            for config := range planner.Plan(task.Spec) {
                select {
                case configChan <- config:
                case <-stopChan:
                    return
                }
            }
        case <-stopChan:
            return
        }
    }
}
```

### 3.2 Packet Worker

**职责**：消费配置，构造报文

```go
func packetWorker(
    id int,
    configChan <-chan PacketConfig,
    packetChan chan<- []byte,
    rateLimiters map[string]*RateLimiter,
    errChan chan<- error,
    stopChan <-chan struct{},
) {
    defer func() {
        if r := recover(); r != nil {
            errChan <- fmt.Errorf("packet worker %d panic: %v", id, r)
        }
    }()
    
    for {
        select {
        case config, ok := <-configChan:
            if !ok {
                return
            }
            
            // 速率限制
            if limiter, ok := rateLimiters[config.ClassID]; ok {
                if err := limiter.Wait(len(config.Payload)); err != nil {
                    errChan <- fmt.Errorf("rate limit error: %w", err)
                    continue
                }
            }
            
            // 构造报文
            packet, err := builders.Build(config)
            if err != nil {
                errChan <- fmt.Errorf("build packet error: %w", err)
                continue
            }
            
            select {
            case packetChan <- packet:
            case <-stopChan:
                return
            }
            
        case <-stopChan:
            return
        }
    }
}
```

## 4. 环形缓冲区

### 4.1 设计目标

- 有界容量，防止内存溢出
- 线程安全
- 高性能读写

### 4.2 数据结构

```go
type RingBuffer struct {
    buffer   []Packet
    capacity int
    size     int
    head     int
    tail     int
    mu       sync.RWMutex
}

func NewRingBuffer(capacity int) *RingBuffer {
    return &RingBuffer{
        buffer:   make([]Packet, capacity),
        capacity: capacity,
    }
}

func (rb *RingBuffer) Put(packet Packet) bool {
    rb.mu.Lock()
    defer rb.mu.Unlock()
    
    if rb.size >= rb.capacity {
        return false // 缓冲区满
    }
    
    rb.buffer[rb.tail] = packet
    rb.tail = (rb.tail + 1) % rb.capacity
    rb.size++
    return true
}

func (rb *RingBuffer) GetBatch(count int) []Packet {
    rb.mu.Lock()
    defer rb.mu.Unlock()
    
    n := min(count, rb.size)
    result := make([]Packet, n)
    
    for i := 0; i < n; i++ {
        result[i] = rb.buffer[rb.head]
        rb.head = (rb.head + 1) % rb.capacity
    }
    
    rb.size -= n
    return result
}
```

## 5. 速率限制器

### 5.1 令牌桶算法

```go
type RateLimiter struct {
    rate     float64 // 字节/秒
    capacity int64   // 桶容量
    tokens   int64   // 当前令牌数
    lastTime time.Time
    mu       sync.Mutex
}

func (rl *RateLimiter) Wait(bytes int) {
    rl.mu.Lock()
    defer rl.mu.Unlock()
    
    // 补充令牌
    now := time.Now()
    elapsed := now.Sub(rl.lastTime).Seconds()
    rl.tokens = min(rl.capacity, 
        rl.tokens + int64(elapsed*rl.rate))
    rl.lastTime = now
    
    // 等待令牌
    for rl.tokens < int64(bytes) {
        rl.mu.Unlock()
        time.Sleep(time.Millisecond)
        rl.mu.Lock()
        
        now = time.Now()
        elapsed = now.Sub(rl.lastTime).Seconds()
        rl.tokens = min(rl.capacity, 
            rl.tokens + int64(elapsed*rl.rate))
        rl.lastTime = now
    }
    
    rl.tokens -= int64(bytes)
}
```
