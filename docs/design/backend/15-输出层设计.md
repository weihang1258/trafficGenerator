## 3. 网卡输出实现

```go
// pkg/output/interface_writer.go
import "github.com/google/gopacket/pcap"

type InterfaceWriter struct {
    handle    *pcap.Handle
    ifaceName string
    stats     OutputStats
    mu        sync.Mutex
}

func NewInterfaceWriter(ifaceName string) (*InterfaceWriter, error) {
    handle, err := pcap.OpenLive(
        ifaceName,
        65536,  // snaplen
        true,   // promiscuous
        pcap.BlockForever,
    )
    if err != nil {
        return nil, err
    }
    
    return &InterfaceWriter{
        handle:    handle,
        ifaceName: ifaceName,
    }, nil
}

func (w *InterfaceWriter) Write(packet []byte) error {
    if err := w.handle.WritePacketData(packet); err != nil {
        atomic.AddInt64(&w.stats.Errors, 1)
        return err
    }
    
    atomic.AddInt64(&w.stats.PacketsWritten, 1)
    atomic.AddInt64(&w.stats.BytesWritten, int64(len(packet)))
    return nil
}

func (w *InterfaceWriter) WriteBatch(packets [][]byte) error {
    for _, pkt := range packets {
        if err := w.Write(pkt); err != nil {
            return err
        }
    }
    return nil
}

func (w *InterfaceWriter) Close() error {
    w.handle.Close()
    return nil
}
```

## 4. PCAP文件输出

```go
// pkg/output/pcap_writer.go
type PcapWriter struct {
    writer   *pcapgo.Writer
    file     *os.File
    filename string
    stats    OutputStats
    mu       sync.Mutex
}

func NewPcapWriter(filename string) (*PcapWriter, error) {
    f, err := os.Create(filename)
    if err != nil {
        return nil, err
    }
    
    w := pcapgo.NewWriter(f)
    if err := w.WriteFileHeader(65536, layers.LinkTypeEthernet); err != nil {
        f.Close()
        return nil, err
    }
    
    return &PcapWriter{
        writer:   w,
        file:     f,
        filename: filename,
    }, nil
}

func (w *PcapWriter) Write(packet []byte) error {
    w.mu.Lock()
    defer w.mu.Unlock()
    
    ci := gopacket.CaptureInfo{
        Timestamp:     time.Now(),
        CaptureLength: len(packet),
        Length:        len(packet),
    }
    
    if err := w.writer.WritePacket(ci, packet); err != nil {
        atomic.AddInt64(&w.stats.Errors, 1)
        return err
    }
    
    atomic.AddInt64(&w.stats.PacketsWritten, 1)
    atomic.AddInt64(&w.stats.BytesWritten, int64(len(packet)))
    return nil
}

func (w *PcapWriter) Close() error {
    return w.file.Close()
}
```

## 5. 输出管理器

```go
// pkg/output/manager.go
type OutputManager struct {
    writers []OutputWriter
    mode    string // interface, pcap, both, callback
    mu      sync.RWMutex
}

func NewOutputManager(config OutputConfig) (*OutputManager, error) {
    om := &OutputManager{
        mode: config.Mode,
    }
    
    // 网卡输出
    if config.Mode == "interface" || config.Mode == "both" {
        for _, iface := range config.Interfaces {
            w, err := NewInterfaceWriter(iface)
            if err != nil {
                return nil, err
            }
            om.writers = append(om.writers, w)
        }
    }
    
    // PCAP文件输出
    if config.Mode == "pcap" || config.Mode == "both" {
        w, err := NewPcapWriter(config.PcapFile)
        if err != nil {
            return nil, err
        }
        om.writers = append(om.writers, w)
    }
    
    return om, nil
}

func (om *OutputManager) Write(packet []byte) error {
    om.mu.RLock()
    defer om.mu.RUnlock()
    
    for _, w := range om.writers {
        if err := w.Write(packet); err != nil {
            log.Error("write failed", zap.Error(err))
        }
    }
    return nil
}
```

## 7. 与核心引擎集成

```go
// pkg/core/engine.go
type Engine struct {
    // ... 现有字段
    outputMgr *output.OutputManager
}

func (e *Engine) Start() error {
    // ... 现有启动逻辑
    
    // 启动输出Worker
    e.wg.Add(1)
    go e.outputWorker()
    
    return nil
}

func (e *Engine) outputWorker() {
    defer e.wg.Done()
    
    ticker := time.NewTicker(10 * time.Millisecond)
    defer ticker.Stop()
    
    for {
        select {
        case <-ticker.C:
            // 从缓冲区批量获取
            packets := e.combinedBuffer.GetBatch(100)
            if len(packets) == 0 {
                continue
            }
            
            // 批量输出
            for _, pkt := range packets {
                e.outputMgr.Write(pkt.Data)
            }
            
        case <-e.shutdownCh:
            return
        }
    }
}
```

## 8. 使用示例

engine := core.NewEngine(&core.EngineConfig{
    Output: output.OutputConfig{
        Mode:       "interface",
        Interfaces: []string{"eth0"},
    },
})

// PCAP文件输出
engine := core.NewEngine(&core.EngineConfig{
    Output: output.OutputConfig{
        Mode:     "pcap",
        PcapFile: "/tmp/traffic.pcap",
    },

// 同时输出到网卡和文件
engine := core.NewEngine(&core.EngineConfig{
    Output: output.OutputConfig{
        Mode:       "both",
        Interfaces: []string{"eth0"},
        PcapFile:   "/tmp/traffic.pcap",
    },
})
