## 1. 网卡管理器

### 1.1 网卡发现

```go
// pkg/netif/manager.go
package netif

import (
    "net"
    "github.com/google/gopacket/pcap"
)

type InterfaceInfo struct {
    Name       string   `json:"name"`
    MAC        string   `json:"mac"`
    IPs        []string `json:"ips"`
    LinkStatus bool     `json:"link_status"`
    Speed      int      `json:"speed"` // Mbps
    MTU        int      `json:"mtu"`
}

type InterfaceManager struct {
    interfaces map[string]*InterfaceInfo
    mu         sync.RWMutex
}

func NewInterfaceManager() *InterfaceManager {
    return &InterfaceManager{
        interfaces: make(map[string]*InterfaceInfo),
    }
}

// 自动发现所有网卡
func (im *InterfaceManager) DiscoverInterfaces() error {
    devices, err := pcap.FindAllDevs()
    if err != nil {
        return err
    }
    
    im.mu.Lock()
    defer im.mu.Unlock()
    
    for _, dev := range devices {
        if dev.Name == "lo" || dev.Name == "any" {
            continue // 跳过回环接口
        }
        
        info := &InterfaceInfo{
            Name:       dev.Name,
            LinkStatus: im.checkLinkStatus(dev.Name),
            MTU:        im.getMTU(dev.Name),
        }
        
        // 获取MAC和IP
        if iface, err := net.InterfaceByName(dev.Name); err == nil {
            info.MAC = iface.HardwareAddr.String()
            if addrs, err := iface.Addrs(); err == nil {
                for _, addr := range addrs {
                    info.IPs = append(info.IPs, addr.String())
                }
            }
        }
        
        im.interfaces[dev.Name] = info
    }
    
    return nil
}

// 检查网卡Link状态
func (im *InterfaceManager) checkLinkStatus(name string) bool {
    iface, err := net.InterfaceByName(name)
    if err != nil {
        return false
    }
    return iface.Flags&net.FlagUp != 0
}

// 获取可用网卡列表
func (im *InterfaceManager) GetAvailableInterfaces() []InterfaceInfo {
    im.mu.RLock()
    defer im.mu.RUnlock()
    
    var result []InterfaceInfo
    for _, info := range im.interfaces {
        if info.LinkStatus {
            result = append(result, *info)
        }
    }
    return result
}

// 验证网卡是否可用
func (im *InterfaceManager) ValidateInterface(name string) error {
    im.mu.RLock()
    defer im.mu.RUnlock()
    
    info, exists := im.interfaces[name]
    if !exists {
        return fmt.Errorf("interface %s not found", name)
    }
    
    if !info.LinkStatus {
        return fmt.Errorf("interface %s link is down", name)
    }
    
    return nil
}
```

## 2. 端口调度器

### 2.1 端口资源管理

```go
// pkg/netif/port_scheduler.go
type PortScheduler struct {
    // 端口占用表: interface -> port -> taskID
    allocations map[string]map[uint16]string
    // 等待队列: interface -> []waitingTask
    waitQueue   map[string][]*WaitingTask
    mu          sync.RWMutex
    cond        *sync.Cond
}

type WaitingTask struct {
    TaskID    string
    Interface string
    Port      uint16
    Timeout   time.Time
    ResultCh  chan error
}

func NewPortScheduler() *PortScheduler {
    ps := &PortScheduler{
        allocations: make(map[string]map[uint16]string),
        waitQueue:   make(map[string][]*WaitingTask),
    }
    ps.cond = sync.NewCond(&ps.mu)
    return ps
}

// 分配端口（阻塞直到可用）
func (ps *PortScheduler) AllocatePort(taskID, iface string, port uint16, timeout time.Duration) error {
    ps.mu.Lock()
    
    // 检查端口是否已被占用
    if ports, exists := ps.allocations[iface]; exists {
        if existingTask, occupied := ports[port]; occupied {
            if existingTask == taskID {
                ps.mu.Unlock()
                return nil // 已经分配给当前任务
            }
            
            // 端口被占用，加入等待队列
            waiting := &WaitingTask{
                TaskID:    taskID,
                Interface: iface,
                Port:      port,
                Timeout:   time.Now().Add(timeout),
                ResultCh:  make(chan error, 1),
            }
            
            ps.waitQueue[iface] = append(ps.waitQueue[iface], waiting)
            ps.mu.Unlock()
            
            // 等待端口释放或超时
            select {
            case err := <-waiting.ResultCh:
                return err
            case <-time.After(timeout):
                ps.removeFromWaitQueue(taskID, iface, port)
                return fmt.Errorf("port allocation timeout")
            }
        }
    }
    
    // 端口可用，立即分配
    if ps.allocations[iface] == nil {
        ps.allocations[iface] = make(map[uint16]string)
    }
    ps.allocations[iface][port] = taskID
    ps.mu.Unlock()
    
    log.Info("port allocated", 
        zap.String("task", taskID),
        zap.String("interface", iface),
        zap.Uint16("port", port))
    
    return nil
}

// 释放端口
func (ps *PortScheduler) ReleasePort(taskID, iface string, port uint16) {
    ps.mu.Lock()
    defer ps.mu.Unlock()
    
    if ports, exists := ps.allocations[iface]; exists {
        if ports[port] == taskID {
            delete(ports, port)
            log.Info("port released",
                zap.String("task", taskID),
                zap.String("interface", iface),
                zap.Uint16("port", port))
            
            // 通知等待队列
            ps.notifyWaitQueue(iface, port)
        }
    }
}

// 通知等待队列
func (ps *PortScheduler) notifyWaitQueue(iface string, port uint16) {
    queue := ps.waitQueue[iface]
    for i, waiting := range queue {
        if waiting.Port == port && time.Now().Before(waiting.Timeout) {
            // 分配给等待的任务
            if ps.allocations[iface] == nil {
                ps.allocations[iface] = make(map[uint16]string)
            }
            ps.allocations[iface][port] = waiting.TaskID
            
            // 从队列移除并通知
            ps.waitQueue[iface] = append(queue[:i], queue[i+1:]...)
            waiting.ResultCh <- nil
            return
        }
    }
}

// 批量释放任务的所有端口
func (ps *PortScheduler) ReleaseTaskPorts(taskID string) {
    ps.mu.Lock()
    defer ps.mu.Unlock()
    
    for iface, ports := range ps.allocations {
        for port, tid := range ports {
            if tid == taskID {
                delete(ports, port)
                ps.notifyWaitQueue(iface, port)
            }
        }
    }
}
```

## 3. 任务启动前检查

```go
// pkg/core/engine.go
func (e *Engine) SubmitTask(task Task) error {
    // 1. 检查网卡是否存在且Link状态正常
    if err := e.ifaceMgr.ValidateInterface(task.Interface); err != nil {
        return fmt.Errorf("interface validation failed: %w", err)
    }
    
    // 2. 分配端口（阻塞等待或超时）
    if err := e.portScheduler.AllocatePort(
        task.ID, 
        task.Interface, 
        task.Spec.SrcPort,
        30*time.Second, // 等待超时
    ); err != nil {
        return fmt.Errorf("port allocation failed: %w", err)
    }
    
    // 3. 提交任务到队列
    select {
    case e.taskChan <- task:
        return nil
    case <-time.After(5 * time.Second):
        e.portScheduler.ReleasePort(task.ID, task.Interface, task.Spec.SrcPort)
        return fmt.Errorf("task queue full")
    }
}

// 任务完成后自动释放端口
func (e *Engine) onTaskComplete(taskID string) {
    e.portScheduler.ReleaseTaskPorts(taskID)
}
```

```go
// internal/api/rest/interface.go

// GET /api/v1/interfaces
func listInterfaces(c *gin.Context) {
    interfaces := ifaceMgr.GetAvailableInterfaces()
    c.JSON(200, ApiResponse{
        Code: 0,
        Data: interfaces,
    })
}

// POST /api/v1/interfaces/discover
func discoverInterfaces(c *gin.Context) {
    if err := ifaceMgr.DiscoverInterfaces(); err != nil {
        c.JSON(500, ApiResponse{
            Code:    500,
            Message: err.Error(),
        })
        return
    }
    c.JSON(200, ApiResponse{Code: 0, Message: "success"})
}

// GET /api/v1/ports/status
func getPortStatus(c *gin.Context) {
    status := portScheduler.GetAllocationStatus()
    c.JSON(200, ApiResponse{
        Code: 0,
        Data: status,
    })
}
```

```vue
import { interfaceApi } from '@/api/interface'

const discovering = ref(false)

  loading.value = true
  try {
    interfaces.value = await interfaceApi.list()
  } finally {
    loading.value = false

  discovering.value = true
  try {

    await fetchInterfaces()
    ElMessage.success('网卡发现完成')
  } finally {
    discovering.value = false
  }
}
```
