## 1. 构造流程

```
L2 (Ethernet) → L3 (IPv4) → L4 (TCP/UDP) → Payload
```

---

## 2. L2层构造

```go
// pkg/builders/ethernet.go
func BuildEthernet(src, dst string, vlan *VLAN) []byte {
    frame := make([]byte, 14)
    copy(frame[0:6], parseMAC(dst))
    copy(frame[6:12], parseMAC(src))
    
    if vlan != nil {
        // 802.1Q
        binary.BigEndian.PutUint16(frame[12:14], 0x8100)
        // TCI + EtherType
    } else {
        binary.BigEndian.PutUint16(frame[12:14], 0x0800) // IPv4
    }
    
    return frame
}
```

## 3. L3层构造

```go
// pkg/builders/ipv4.go
func BuildIPv4(src, dst string, proto uint8, payload []byte) []byte {
    header := make([]byte, 20)
    header[0] = 0x45 // Version + IHL
    totalLen := 20 + len(payload)
    binary.BigEndian.PutUint16(header[2:4], uint16(totalLen))
    header[9] = proto
    copy(header[12:16], net.ParseIP(src).To4())
    copy(header[16:20], net.ParseIP(dst).To4())
    
    // Checksum
    checksum := calculateChecksum(header)
    binary.BigEndian.PutUint16(header[10:12], checksum)
    
    return append(header, payload...)
}
```

## 4. L4层构造

### TCP

```go
// pkg/builders/tcp.go
func BuildTCP(src, dst uint16, seq, ack uint32, flags uint8, payload []byte) []byte {
    header := make([]byte, 20)
    binary.BigEndian.PutUint16(header[0:2], src)
    binary.BigEndian.PutUint16(header[2:4], dst)
    binary.BigEndian.PutUint32(header[4:8], seq)
    binary.BigEndian.PutUint32(header[8:12], ack)
    header[12] = 0x50 // Data offset
    header[13] = flags
    binary.BigEndian.PutUint16(header[14:16], 65535) // Window
    
    return append(header, payload...)
}
```

### UDP

```go
// pkg/builders/udp.go
func BuildUDP(src, dst uint16, payload []byte) []byte {
    header := make([]byte, 8)
    binary.BigEndian.PutUint16(header[0:2], src)
    binary.BigEndian.PutUint16(header[2:4], dst)
    length := uint16(8 + len(payload))
    binary.BigEndian.PutUint16(header[4:6], length)
    
    return append(header, payload...)
}
```

## 5. 报文组装

```go
// pkg/builders/assembler.go
func Assemble(config PacketConfig) []byte {
    // L4
    var l4 []byte
    if config.L4.Protocol == "tcp" {
        l4 = BuildTCP(config.L4.SrcPort, config.L4.DstPort, 
            config.L4.Seq, config.L4.Ack, config.L4.Flags, config.Payload)
    } else {
        l4 = BuildUDP(config.L4.SrcPort, config.L4.DstPort, config.Payload)
    }
    
    // L3
    l3 := BuildIPv4(config.L3.SrcIP, config.L3.DstIP, 
        getProtoNum(config.L4.Protocol), l4)
    
    // L2
    l2 := BuildEthernet(config.L2.SrcMAC, config.L2.DstMAC, config.L2.VLAN)
    
    return append(l2, l3...)
}
```
