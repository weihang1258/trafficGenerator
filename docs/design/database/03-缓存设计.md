## 1. Redis缓存

### 缓存键设计

```
task:status:{task_id}     # 任务状态
task:stats:{task_id}      # 任务统计
system:status             # 系统状态
strategy:list             # 策略列表
```

---

## 2. 缓存策略

| 数据 | TTL | 更新策略 |
|------|-----|---------|
| 任务状态 | 60s | 实时更新 |
| 任务统计 | 10s | 定时更新 |
| 系统状态 | 5s | 定时更新 |
| 策略列表 | 300s | 变更时清除 |

---

## 3. 实现

```go
// internal/cache/redis.go
import "github.com/go-redis/redis/v8"

type Cache struct {
    client *redis.Client
}

func (c *Cache) SetTaskStatus(taskID string, status TaskStatus) error {
    key := fmt.Sprintf("task:status:%s", taskID)
    data, _ := json.Marshal(status)
    return c.client.Set(ctx, key, data, 60*time.Second).Err()
}

func (c *Cache) GetTaskStatus(taskID string) (*TaskStatus, error) {
    key := fmt.Sprintf("task:status:%s", taskID)
    data, err := c.client.Get(ctx, key).Bytes()
    if err != nil {
        return nil, err
    }
    
    var status TaskStatus
    json.Unmarshal(data, &status)
    return &status, nil
}
```

## 4. 缓存更新

```go
// 写入时更新缓存
func (s *TaskService) UpdateTaskStatus(taskID string, status string) error {
    // 更新数据库
    err := s.repo.UpdateStatus(taskID, status)
    if err != nil {
        return err
    }
    
    // 更新缓存
    s.cache.SetTaskStatus(taskID, TaskStatus{
        TaskID: taskID,
        Status: status,
    })
    
    return nil
}
```
