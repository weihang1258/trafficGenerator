## 3. 初始化迁移

### 000001_init_schema.up.sql

```sql
-- 任务表
CREATE TABLE IF NOT EXISTS tasks (
    id VARCHAR(64) PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    spec JSON NOT NULL,
    status VARCHAR(32) NOT NULL DEFAULT 'created',
    progress DECIMAL(5,2) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    started_at TIMESTAMP NULL,
    completed_at TIMESTAMP NULL,
    INDEX idx_status (status),
    INDEX idx_created_at (created_at)
);

-- 策略表
CREATE TABLE IF NOT EXISTS strategies (
    id VARCHAR(64) PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    protocol VARCHAR(32) NOT NULL,
    description TEXT,
    config JSON NOT NULL,
    usage_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_protocol (protocol),
    INDEX idx_name (name)
);

-- 统计表
CREATE TABLE IF NOT EXISTS task_stats (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    task_id VARCHAR(64) NOT NULL,
    packets_sent BIGINT DEFAULT 0,
    bytes_sent BIGINT DEFAULT 0,
    flows_generated INT DEFAULT 0,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE,
    INDEX idx_task_timestamp (task_id, timestamp)
);
```

### 000001_init_schema.down.sql

```sql
DROP TABLE IF EXISTS task_stats;
DROP TABLE IF EXISTS strategies;
DROP TABLE IF EXISTS tasks;
```

## 4. 迁移管理代码

```go
// internal/storage/migrate.go
package storage

import (
    "database/sql"
    "embed"
    
    "github.com/golang-migrate/migrate/v4"
    "github.com/golang-migrate/migrate/v4/database/sqlite3"
    "github.com/golang-migrate/migrate/v4/source/iofs"
)

//go:embed migrations/*.sql
var migrationsFS embed.FS

func RunMigrations(db *sql.DB) error {
    driver, err := sqlite3.WithInstance(db, &sqlite3.Config{})
    if err != nil {
        return err
    }
    
    source, err := iofs.New(migrationsFS, "migrations")
    if err != nil {
        return err
    }
    
    m, err := migrate.NewWithInstance("iofs", source, "sqlite3", driver)
    if err != nil {
        return err
    }
    
    if err := m.Up(); err != nil && err != migrate.ErrNoChange {
        return err
    }
    
    return nil
}
```

## 5. 使用方式

### 命令行

```bash
# 创建新迁移
migrate create -ext sql -dir internal/storage/migrations -seq add_user_table

# 执行迁移
migrate -path internal/storage/migrations -database "sqlite3://./data.db" up

# 回滚
migrate -path internal/storage/migrations -database "sqlite3://./data.db" down 1

# 查看版本
migrate -path internal/storage/migrations -database "sqlite3://./data.db" version
```

### 代码中执行

```go
// cmd/server/main.go
db, err := sql.Open("sqlite3", "./data.db")
if err != nil {
    log.Fatal(err)
}

if err := storage.RunMigrations(db); err != nil {
    log.Fatal("migration failed:", err)
}
```

## 7. 迁移最佳实践

1. **向后兼容**: 新迁移不破坏旧数据
2. **可回滚**: 每个up都有对应down
3. **原子性**: 使用事务
4. **测试**: 在测试环境验证
5. **备份**: 生产环境迁移前备份
